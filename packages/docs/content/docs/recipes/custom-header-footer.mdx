# Custom Header & Footer

Use `header` and `footer` props to add custom contextual information.

<HeaderFooterDemo />

---

## Code Example

```tsx
import { useState } from 'react';
import { Calendar } from 'react-modular-datepicker';

export function HeaderFooterExample() {
  const [selected, setSelected] = useState<Date | null>(new Date());

  return (
    <Calendar
      selected={selected}
      onChange={(val) => setSelected(val as Date)}
      header={<div className="p-2 bg-blue-100 text-blue-800 font-bold text-center">Header Banner</div>}
      footer={<div className="p-2 bg-gray-100 text-gray-600 text-xs text-center">Footer text</div>}
    />
  );
}
```
