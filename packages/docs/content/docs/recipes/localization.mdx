# Localization & RTL Support

Customize the starting day of the week with `firstDayOfWeek`, pass localized month and weekday arrays via `translations`, and support Right-To-Left (RTL) languages like Arabic or Hebrew by placing the calendar inside a container with `dir="rtl"`.

<LocalizationDemo />

---

## Code Example

### Spanish (LTR)

```tsx
import { Calendar } from 'react-modular-datepicker';

export function SpanishCalendarExample() {
  return (
    <Calendar
      firstDayOfWeek={1} // Start week on Monday
      translations={{
        months: [
          'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio',
          'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'
        ],
        weekdays: ['Dom', 'Lun', 'Mar', 'Mié', 'Jue', 'Vie', 'Sáb'],
        back: 'Mes anterior',
        forward: 'Mes siguiente',
      }}
    />
  );
}
```

### Arabic (RTL)

```tsx
import { Calendar } from 'react-modular-datepicker';

export function ArabicRtlCalendarExample() {
  return (
    <div dir="rtl">
      <Calendar
        firstDayOfWeek={6} // Start week on Saturday
        translations={{
          months: [
            'يناير', 'فبراير', 'مارس', 'أبريل', 'مايو', 'يونيو',
            'يوليو', 'أغسطس', 'سبتمبر', 'أكتوبر', 'نوفمبر', 'ديسمبر'
          ],
          weekdays: ['الأحد', 'الإثنين', 'الثلاثاء', 'الأربعاء', 'الخميس', 'الجمعة', 'السبت'],
          back: 'الشهر السابق',
          forward: 'الشهر التالي',
        }}
      />
    </div>
  );
}
```
