# Availability & Booking Calendar

Create availability calendars for appointments, hotel bookings, or resource scheduling using disabled dates and custom modifier classes.

## Code Example

```tsx
import { useState } from 'react';
import { Calendar } from 'react-modular-datepicker';
import 'react-modular-datepicker/dist/index.css';

const bookedDays = [
  new Date(2025, 2, 3),
  new Date(2025, 2, 4),
  new Date(2025, 2, 10),
];

const availableSlotsMap: Record<string, string[]> = {
  '2025-03-08': ['09:00 AM', '10:30 AM', '02:00 PM'],
  '2025-03-09': ['01:00 PM', '03:30 PM'],
  '2025-03-15': ['09:00 AM', '11:00 AM', '04:00 PM'],
};

export function AvailabilityExample() {
  const [selectedDate, setSelectedDate] = useState<Date | null>(new Date(2025, 2, 8));
  const [selectedSlot, setSelectedSlot] = useState<string | null>(null);

  const getKey = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
  const slots = selectedDate ? availableSlotsMap[getKey(selectedDate)] || [] : [];

  return (
    <div className="flex flex-col md:flex-row gap-6">
      <Calendar
        selected={selectedDate || undefined}
        onChange={(d) => {
          setSelectedDate(d as Date);
          setSelectedSlot(null);
        }}
        disabledDates={bookedDays}
        modifiers={{
          available: (d) => !!availableSlotsMap[getKey(d)],
        }}
        classNames={{
          day: {
            disabled: 'bg-red-50 text-red-300 line-through cursor-not-allowed',
            available: 'font-semibold text-emerald-600',
          },
        }}
      />

      <div className="flex-1 p-4 border rounded-xl">
        <h3 className="font-bold text-lg mb-3">
          {selectedDate ? `Slots for ${selectedDate.toLocaleDateString()}` : 'Select a date'}
        </h3>
        {slots.length === 0 ? (
          <p className="text-sm text-gray-500">No available time slots.</p>
        ) : (
          <div className="grid grid-cols-2 gap-2">
            {slots.map((slot) => (
              <button
                key={slot}
                onClick={() => setSelectedSlot(slot)}
                className={`py-2 px-3 text-xs font-semibold border rounded-lg ${
                  selectedSlot === slot ? 'bg-emerald-600 text-white' : 'bg-gray-50'
                }`}
              >
                {slot}
              </button>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
```
