# Headless Usage

The `useDates` hook provides complete layout flexibility while handling date math, selection state, and accessibility attributes.

<HeadlessDemo />

---

## Code Example

```tsx
import { useState } from 'react';
import { useDates } from 'react-modular-datepicker';

export function HeadlessExample() {
  const [selectedDate, setSelectedDate] = useState<Date | undefined>(new Date());

  const { calendars, getBackProps, getForwardProps, getDateProps } = useDates({
    selected: selectedDate,
    onChange: (d) => setSelectedDate(d as Date),
  });

  const calendar = calendars[0];

  return (
    <div className="border p-4 rounded-xl max-w-sm">
      <div className="flex justify-between items-center mb-4">
        <button {...getBackProps()} className="px-2 py-1 bg-gray-100 rounded">
          ←
        </button>
        <span className="font-bold">
          {calendar.year} - {calendar.month + 1}
        </span>
        <button {...getForwardProps()} className="px-2 py-1 bg-gray-100 rounded">
          →
        </button>
      </div>

      <div className="grid grid-cols-7 gap-1">
        {calendar.weeks.flat().map((dateObj, idx) => (
          dateObj ? (
            <button
              key={idx}
              {...getDateProps({ dateObj })}
              className={`h-8 w-8 rounded-full ${
                dateObj.selected ? 'bg-blue-600 text-white font-bold' : 'hover:bg-gray-100'
              }`}
            >
              {dateObj.date.getDate()}
            </button>
          ) : <div key={idx} />
        ))}
      </div>
    </div>
  );
}
```
