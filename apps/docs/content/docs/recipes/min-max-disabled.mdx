# Min, Max & Disabled Dates

Restrict selection with `minDate`, `maxDate`, and `disabledDates`.

<MinMaxDisabledDemo />

---

## Code Example

```tsx
import { useState } from 'react';
import { Calendar } from 'react-modular-datepicker';

export function BoundsExample() {
  const [selected, setSelected] = useState<Date | null>(new Date());

  const today = new Date();
  const minDate = new Date(today.getFullYear(), today.getMonth(), 5);
  const maxDate = new Date(today.getFullYear(), today.getMonth(), 25);
  const disabledDates = [
    new Date(today.getFullYear(), today.getMonth(), 10),
    new Date(today.getFullYear(), today.getMonth(), 12),
  ];

  return (
    <Calendar
      selected={selected}
      onChange={(d) => setSelected(d as Date)}
      minDate={minDate}
      maxDate={maxDate}
      disabledDates={disabledDates}
    />
  );
}
```
